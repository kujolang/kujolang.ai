# TOML parsing and generation

let toml_text := "title = \"Kujo\"\nversion = \"1.0.0\"\n[server]\nhost = \"127.0.0.1\"\nport = 8080"
let config := parse_toml(toml_text)

assert_equal(config["title"], "Kujo")
assert_equal(config["server"]["port"], 8080)

let generated := to_toml(config)
let round_trip := parse_toml(generated)

assert_equal(round_trip["server"]["host"], "127.0.0.1")
print("parsed TOML:", config)
