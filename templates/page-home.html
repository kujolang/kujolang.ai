Functions and operators

Add the numbers in an array with a function.

# Expected output:
# total: 42

func total(values) {
    mut sum := 0
    for value in values {
        sum = sum + value
    }
    return sum
}

let scores := [8, 13, 21]
print("total: " + to_string(total(scores)))

Save as functions.kujo, then:

kujo run functions.kujo
Expected output
total: 42

AI runtime · Executed offline

Deterministic AI request IDs

Hash an AI request without including credentials.

The credentials are dummy values. This example makes no network request.

func kv(label, value) {
    print(label + ": " + value)
}

let options_a := {
    "endpoint": "https://api.example.test/v1/chat/completions",
    "model": "gpt-demo",
    "api_key": "credential-one",
    "headers": {
        "Authorization": "Bearer credential-one",
        "X-App": "kujo"
    },
    "body": {
        "temperature": 0,
        "max_tokens": 32
    }
}

let options_b := {
    "endpoint": "https://api.example.test/v1/chat/completions",
    "model": "gpt-demo",
    "api_key": "credential-two",
    "headers": {
        "Authorization": "Bearer credential-two",
        "X-App": "kujo"
    },
    "body": {
        "temperature": 0,
        "max_tokens": 32
    }
}

let hash_a := ai_request_hash("Summarize Kujo", options_a)
let hash_b := ai_request_hash("Summarize Kujo", options_b)

kv("hash_length", to_string(len(hash_a)))
kv("same_without_credentials", to_string(hash_a == hash_b))
kv("hash", hash_a)

Save as request-hash.kujo, then:

kujo run request-hash.kujo
Expected output
hash_length: 64
same_without_credentials: true
hash: a6244047c108dcf58a952ed0a6c70dc04d13ee9aa9dba08eeb623f241fbd37ce

AI calls and replay

Replay a recorded model response offline.

Use the samples bundle or the pinned Kujo checkout so the cassette fixtures are available.

func kv(label, value) {
    print(label + ": " + value)
}

let options := {
    "endpoint": "http://127.0.0.1:1/v1/chat/completions",
    "model": "gpt-replay",
    "cassette": {
        "mode": "replay",
        "dir": "tests/fixtures/ai_cassettes"
    }
}

let result := ai_chat("Hello model", options)

kv("replay_result", to_string(result))

Save as replay.kujo, then:

kujo run replay.kujo
Expected output
