# Create HTTP server on port 8080
server := http_server(8080)

# Add a simple GET route
server := server.route("GET", "/", func(req) {
    return http_response(200, "Hello, World!")
})

# Add another route
server := server.route("GET", "/hello", func(req) {
    return http_response(200, "Hello from Kujo!")
})

# Start the server (blocking - will run until Ctrl+C)
server.listen()
