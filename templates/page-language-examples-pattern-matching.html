# Pattern matching examples using current Kujo match syntax.

enum Status {
    Success,
    Failure
}

func status_for(code) {
    if code == 200 {
        return Status::Success("ok")
    }
    return Status::Failure("not ok")
}

let result := status_for(200)

match result {
    case Status::Success(message): {
        print("success: " + message)
    }
    case Status::Failure(error): {
        print("failure: " + error)
    }
}

let maybe_value := Ok(42)

match maybe_value {
    case Ok(value): {
        print("value: " + to_string(value))
    }
    case Err(message): {
        print("error: " + message)
    }
}
