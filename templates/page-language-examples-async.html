Async tasks

Await a value produced by an async function.

// Simple async test

async func get_value() {
    return 42;
}

let p = get_value();
println("Promise type: ", type(p));
let result = await p;
println("Result: ", result);

Save as async.kujo, then:

kujo run async.kujo
Expected output
Promise type:  promise
Result:  42
