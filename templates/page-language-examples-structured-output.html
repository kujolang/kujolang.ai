Structured output validation

Validate structured data before consuming it. This validates local values; it does not call a model.

func kv(label, value) {
    print(label + ": " + to_string(value))
}

let schema := {
    "type": "object",
    "required": ["name", "scores"],
    "properties": {
        "name": {
            "type": "string",
            "minLength": 2
        },
        "scores": {
            "type": "array",
            "minItems": 1,
            "items": {
                "type": "number",
                "minimum": 0,
                "maximum": 100
            }
        }
    },
    "additionalProperties": false
}

let valid_report := json_schema_validate({
    "name": "Kujo",
    "scores": [95, 88]
}, schema)

let invalid_report := json_schema_validate({
    "name": "K",
    "scores": [101],
    "debug": true
}, schema)

kv("valid", valid_report["valid"])
kv("invalid", invalid_report["valid"])
kv("errors", invalid_report["errors"])

Save as structured-output.kujo, then:

kujo run structured-output.kujo
Expected output
valid: true
invalid: false
errors: [{"keyword": minLength, "message": string length is less than 2, "path": /name}, {"keyword": maximum, "message": number is greater than 100, "path": /scores/0}, {"keyword": additionalProperties, "message": additional property 'debug' is not allowed, "path": /debug}]
