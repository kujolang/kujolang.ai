# File Operations Demo
# Demonstrates file operation helpers with local output helpers.

func section(title) {
    print("")
    print("== " + title + " ==")
}

func kv(label, value) {
    print("  " + label + ": " + to_string(value))
}

func ok(message) {
    print("  ✓ " + message)
}

section("File Operations Demo")

# Create a test file
section("1. Creating test file")
write_file("demo_test.txt", "Hello, Kujo!\nThis is a test file.\n")
ok("Created demo_test.txt")

# Check file size
section("2. Checking file size")
size := file_size("demo_test.txt")
kv("File size", "${size} bytes")

# Copy the file
section("3. Copying file")
copy_file("demo_test.txt", "demo_test_copy.txt")
ok("Copied to demo_test_copy.txt")

# Verify both files exist
section("4. Verifying both files exist")
exists1 := file_exists("demo_test.txt")
exists2 := file_exists("demo_test_copy.txt")
kv("Original exists", exists1)
kv("Copy exists", exists2)

# Check sizes match
section("5. Comparing file sizes")
size1 := file_size("demo_test.txt")
size2 := file_size("demo_test_copy.txt")
if size1 == size2 {
    ok("Sizes match: ${size1} bytes")
} else {
    print("   ✗ Size mismatch!")
}

# Rename the original
section("6. Renaming original file")
rename_file("demo_test.txt", "demo_renamed.txt")
ok("Renamed to demo_renamed.txt")

# List current directory to see files
section("7. Files in current directory")
files := list_dir(".")
for file in files {
    if contains(file, "demo_") {
        print("   - ${file}")
    }
}

# Clean up
section("8. Cleaning up")
delete_file("demo_renamed.txt")
delete_file("demo_test_copy.txt")
ok("Deleted all test files")

# Verify cleanup
section("9. Verification")
check1 := file_exists("demo_test.txt")
check2 := file_exists("demo_test_copy.txt")
check3 := file_exists("demo_renamed.txt")

if !check1 && !check2 && !check3 {
    ok("All files successfully removed")
} else {
    print("   ✗ Some files still exist")
}

section("Demo Complete")

# Practical example: Backup system
section("Practical Example: Simple Backup")

# Create a config file
config := "database_host=localhost\ndatabase_port=5432\n"
write_file("app_config.txt", config)
ok("Created app_config.txt (${len(config)} bytes)")

# Create backup
if file_exists("app_config.txt") {
    copy_file("app_config.txt", "app_config.backup.txt")
    size := file_size("app_config.backup.txt")
    ok("Backup created: app_config.backup.txt (${size} bytes)")
}

# Modify the config
new_config := "database_host=prod-server\ndatabase_port=5432\n"
write_file("app_config.txt", new_config)
ok("Modified app_config.txt")

# Compare sizes
original_size := file_size("app_config.txt")
backup_size := file_size("app_config.backup.txt")
kv("Original bytes", original_size)
kv("Backup bytes", backup_size)

# Cleanup
delete_file("app_config.txt")
delete_file("app_config.backup.txt")
ok("Cleanup complete")
